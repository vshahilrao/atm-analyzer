import os import re import tkinter as tk from tkinter import filedialog, messagebox, ttk from datetime import datetime import pandas as pd class CashReconciliationApp: def __init__(self, root): self.root = root self.root.title("ATM Cash Dispense & Log Reconciliation App") self.root.geometry("900x680") # Folder Input Frame frame_input = ttk.LabelFrame(root, text="Select Log Folder", padding=10) frame_input.pack(fill="x", padx=10, pady=10) self.path_entry = ttk.Entry(frame_input, width=65) self.path_entry.pack(side="left", padx=5, expand=True, fill="x") ttk.Button(frame_input, text="Browse", command=self.browse_folder).pack(side="left", padx=5) ttk.Button(frame_input, text="Run & Export Excel", command=self.process_and_export).pack(side="left", padx=5) # Output Results Frame frame_results = ttk.LabelFrame(root, text="Reconciliation Audit Summary", padding=10) frame_results.pack(fill="both", expand=True, padx=10, pady=5) self.text_output = tk.Text(frame_results, wrap="word", font=("Consolas", 10)) self.text_output.pack(fill="both", expand=True) def browse_folder(self): selected_dir = filedialog.askdirectory() if selected_dir: self.path_entry.delete(0, tk.END) self.path_entry.insert(0, selected_dir) def process_and_export(self): log_dir = self.path_entry.get().strip() if not log_dir or not os.path.exists(log_dir): messagebox.showerror("Error", "Please select a valid folder path containing log files.") return # Prompt for Excel output path excel_path = filedialog.asksaveasfilename( defaultextension=".xlsx", filetypes=[("Excel Workbook", "*.xlsx"), ("All Files", "*.*")], title="Save Cash Reconciliation Report As..." ) if not excel_path: return total_files = 0 all_transactions = [] all_non_txns = [] # Customizable Regex patterns for ATM Log lines # Example Txn Line: [2026-06-06 10:15:22] TXN_ID: 5021 | Dispensed: 2000 | Taken: 2000 | Purge: 0 | Status: SUCCESS txn_pattern = re.compile( r'(?P\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}).*?' r'TXN_ID:\s*(?P\w+).*?' r'Dispensed:\s*(?P\d+).*?' r'Taken:\s*(?P\d+).*?' r'Purge:\s*(?P\d+).*?' r'Status:\s*(?P\w+)', re.IGNORECASE ) # Example Non-Txn Line: [2026-06-06 10:20:00] EVENT: CASSETTE_LOW_WARNING non_txn_pattern = re.compile( r'(?P\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}).*?' r'EVENT:\s*(?P.+)', re.IGNORECASE ) for root_dir, _, files in os.walk(log_dir): for file in files: if file.endswith((".log", ".txt", ".csv", ".ej")): total_files += 1 file_path = os.path.join(root_dir, file) try: with open(file_path, "r", encoding="utf-8", errors="ignore") as f: for line in f: match = txn_pattern.search(line) if match: data = match.groupdict() all_transactions.append({ 'File Name': file, 'Timestamp': data['timestamp'], 'Transaction_ID': data['txn_id'], 'Dispensed': int(data['dispensed']), 'Taken': int(data['taken']), 'Purge_Bin': int(data['purge']), 'Status': data['status'] }) else: event_match = non_txn_pattern.search(line) if event_match: edata = event_match.groupdict() all_non_txns.append({ 'File Name': file, 'Timestamp': edata['timestamp'], 'Event_Description': edata['event_desc'].strip() }) except Exception as e: print(f"Error reading {file}: {e}") # Dataframe Conversions & Tally Calculations df_txn = pd.DataFrame(all_transactions) df_nontxn = pd.DataFrame(all_non_txns) matched_count = 0 missing_count = 0 total_dispensed_sum = 0 total_taken_sum = 0 total_purge_sum = 0 total_missed_sum = 0 if not df_txn.empty: # Tally logic: Accounted Cash = Taken + Purge Bin df_txn['Accounted_Cash'] = df_txn['Taken'] + df_txn['Purge_Bin'] df_txn['Missed_Or_Discrepancy'] = df_txn['Dispensed'] - df_txn['Accounted_Cash'] df_txn['Reconciliation_Status'] = df_txn['Missed_Or_Discrepancy'].apply( lambda x: 'Matched' if x == 0 else ('Over-accounted' if x < 0 else 'Cash Missing / Unaccounted') ) total_dispensed_sum = df_txn['Dispensed'].sum() total_taken_sum = df_txn['Taken'].sum() total_purge_sum = df_txn['Purge_Bin'].sum() total_missed_sum = df_txn['Missed_Or_Discrepancy'].sum() matched_count = len(df_txn[df_txn['Reconciliation_Status'] == 'Matched']) missing_count = len(df_txn[df_txn['Reconciliation_Status'] != 'Matched']) # Export to Excel with multiple sheets try: with pd.ExcelWriter(excel_path, engine='openpyxl') as writer: if not df_txn.empty: df_txn.to_excel(writer, sheet_name='Cash Reconciliation', index=False) else: pd.DataFrame({'Message': ['No matching transactions found.']}).to_excel(writer, sheet_name='Cash Reconciliation', index=False) if not df_nontxn.empty: df_nontxn.to_excel(writer, sheet_name='Non-Txn Events', index=False) else: pd.DataFrame({'Message': ['No non-transaction events found.']}).to_excel(writer, sheet_name='Non-Txn Events', index=False) except Exception as e: messagebox.showerror("Export Error", f"Could not write Excel file: {e}") return # Display Summary Report in GUI Window self.text_output.delete("1.0", tk.END) report = ( f"=== ATM CASH RECONCILIATION REPORT ===\n" f"Target Folder: {log_dir}\n" f"Exported Excel: {excel_path}\n" f"Files Processed: {total_files}\n" f"---------------------------------------------------\n" f"1. TRANSACTION CASH TALLY:\n" f" ├── Total Transactions Checked: {len(df_txn)}\n" f" ├── Fully Matched Transactions: {matched_count}\n" f" └── Discrepancies / Missed Notes: {missing_count}\n\n" f"2. CASH NUMBERS BREAKDOWN:\n" f" ├── Total Cash Dispensed: {total_dispensed_sum}\n" f" ├── Total Cash Taken: {total_taken_sum}\n" f" ├── Total in Purge Bin: {total_purge_sum}\n" f" └── Total Missed/Lost: {total_missed_sum}\n\n" f"3. NON-TRANSACTION EVENTS:\n" f" └── Total Logged Events: {len(df_nontxn)}\n" f"---------------------------------------------------\n" f"Audit finished successfully! Multi-tab Excel report created." ) self.text_output.insert(tk.END, report) messagebox.showinfo("Success", f"Reconciliation complete!\nExcel report saved successfully.") if __name__ == "__main__": root = tk.Tk() app = CashReconciliationApp(root) root.mainloop()
